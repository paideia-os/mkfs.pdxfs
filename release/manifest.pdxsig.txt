# mkfs.pdxfs v1.0.0 — release manifest (source form)
#
# This is the human-readable source form of `manifest.pdxsig`. The
# release tool at pkgs.paideia-os mirror-push time (a) enumerates the
# artifact list below, (b) recomputes every BLAKE3-256 hash from the
# repo tree at the v1.0.0 tag, (c) writes the completed manifest into a
# binary `manifest.pdxsig`, and (d) invokes
#   paideia-pq-sign::sign_release_artifact(sk, manifest_path)
# once with the paideia-release-line Ed25519 secret key and once with
# the ML-DSA-65 secret key. The two detached signatures are appended to
# the manifest in the fixed layout below and the resulting
# `manifest.pdxsig` is pushed to
#   https://pkgs.paideia-os/main/mkfs.pdxfs/1.0.0/manifest.pdxsig
# alongside the compiled artifacts.
#
# Format mirrors `libpdx-volume`'s `release/manifest.pdxsig.txt` (this
# org's other R53-wave satellite repo, first through M5) byte-for-byte
# in structure — see that file for the worked template this one
# follows. Every `<BLAKE3-*>` / `<...-KID-*>` / `<...-SIG-*>` slot below
# is a placeholder the release tool fills in from the working tree at
# tag time; NONE are computed by this milestone (see
# `release/RELEASE-1.0.0.md` §3 for why).
#
# Reference:
#   design/paideia-as/v0.20-issue-1025-pq-signing.md         — Hybrid,
#     sign_release_artifact, HybridSignature (Ed25519 || ML-DSA-65).
#   design/02-development-environment.md §1140                — hybrid
#     release-artifact signature rationale (Ed25519 + ML-DSA-65).
#   design/tooling/plan.md §6.3                               — package
#     repository layout, `manifest.pdxsig` per version directory.
#   design/tooling/volume-tooling-ux.md §9.1 (paideia-os)     — M5-001
#     scope (dual-signed release + .pdxdoc + mirror push) for this repo.

# ---------------------------------------------------------------
# Signature block — dual-signed per design/02-development-environment.md
# §1140 (hybrid Ed25519 + ML-DSA-65). Both signatures cover the SAME
# canonical byte-stream: the manifest body above (everything before the
# [signatures] marker line) hashed with BLAKE3-256 and signed via
# paideia-pq-sign::sign_release_artifact. Verification is AND-semantics:
# both signatures MUST verify; failure of either rejects the package.
#
# NOT PERFORMED AT THIS MILESTONE. mkfs.pdxfs itself signs every
# file-target superblock with a PLACEHOLDER all-zero seed at v1.0.0
# (src/format.pdx's own "Seed-key strategy" note) precisely because no
# live release-line seed key lives in this repo — the SAME discipline
# applies one level up, to this manifest's own dual-sign pass. The
# signature block below is left as the documented placeholder value
# `SIGNATURE_PLACEHOLDER_PENDING_LIVE_SIGN` in every slot the real
# dual-sign pass would fill; see `release/RELEASE-1.0.0.md` §3 for the
# full rationale and the flag for main.
# ---------------------------------------------------------------

[signatures]
signer-identity      = paideia-release-line
signer-key-source    = design/02-development-environment.md §1164
                       (hardware-backed in CI — TPM 2.0 on the release
                       runner, or a cloud KMS pending PQ-KMS
                       availability check per that section's TODO).
signature-scheme     = hybrid-ed25519+ml-dsa-65 (paideia-pq-hybrid-v1)
hash-input-algo      = blake3-256
hash-input-domain    = "paideia-release-artifact\x00" || manifest-body-bytes
                       (matches paideia-pq-sign::sign_release_artifact
                       domain separation, crates/paideia-pq-sign/src/
                       release.rs.)
